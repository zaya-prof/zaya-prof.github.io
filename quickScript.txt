Sub CombineNameAndLocation()
    Dim ws As Worksheet
    Dim lastRow As Long
    Dim i As Long
    Dim outputRow As Long
    Dim currentName As String
    Dim currentLocation As String
    Dim cellLabel As String
    
    ' Set the worksheet to the one you are currently looking at
    Set ws = ActiveSheet
    
    ' Find the last row of data in Column A
    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    
    ' Set the starting row for your output (E16)
    outputRow = 16
    
    ' Loop through every row in Column A
    For i = 1 To lastRow
        ' Get the label from Column A and trim extra spaces
        cellLabel = Trim(ws.Cells(i, 1).Value)
        
        ' 1. Check if this row is the Customer's Name
        ' We use InStr to be safe in case there are extra spaces or colons
        If InStr(1, cellLabel, "Customer's Name", vbTextCompare) > 0 Then
            currentName = Trim(ws.Cells(i, 2).Value)
        
        ' 2. Check if this row is the City/State
        ElseIf InStr(1, cellLabel, "City/State", vbTextCompare) > 0 Then
            currentLocation = Trim(ws.Cells(i, 2).Value)
        End If
        
        ' 3. If we have found BOTH a name and a location, print them
        If currentName <> "" And currentLocation <> "" Then
            ' Combine them with a space in the middle
            ws.Range("E" & outputRow).Value = currentName & " " & currentLocation
            
            ' Move the output target down one row for the next person
            outputRow = outputRow + 1
            
            ' Clear the variables so we are ready for the next block of data
            currentName = ""
            currentLocation = ""
        End If
        
    Next i

End Sub
